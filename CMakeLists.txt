cmake_minimum_required(VERSION 3.16)
project(FF2Edit LANGUAGES CXX)

option(FF2EDIT_BUILD_GUI "Build the Qt editor" ${ANDROID})

add_library(ff2save src/save.cpp)
target_include_directories(ff2save PUBLIC include)
target_compile_features(ff2save PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(ff2save PRIVATE /utf-8)
endif()

if(FF2EDIT_BUILD_GUI)
    if(ANDROID)
        # Older Qt Android binaries may have 4 KiB-aligned ELF load segments.
        find_package(Qt6 6.10 REQUIRED COMPONENTS Widgets)
    else()
        find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
    endif()
    set_target_properties(ff2save PROPERTIES POSITION_INDEPENDENT_CODE ON)
    qt_add_executable(ff2edit_gui WIN32 src/gui.cpp)
    target_link_libraries(ff2edit_gui PRIVATE ff2save Qt6::Widgets)
    if(ANDROID)
        target_link_options(ff2edit_gui PRIVATE "-Wl,-z,max-page-size=16384")
    endif()
    if(MSVC)
        target_compile_options(ff2edit_gui PRIVATE /utf-8)
    endif()
endif()

if(NOT ANDROID)
    add_executable(ff2edit src/main.cpp)
    target_link_libraries(ff2edit PRIVATE ff2save)
    if(MINGW)
        target_link_options(ff2edit PRIVATE -municode)
    endif()
endif()
